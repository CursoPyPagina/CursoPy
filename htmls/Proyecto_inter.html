Proyecto intermedio¶

Lo que haremos ahora será diseñar una aplicación que nos permita tener control sobre los miembros en un gimnasio, los ingresos, cuántos clientes al mes tenemos, entre otras cosas. De donde, específicamente, haremos:

  1. Usar la librería de Python tkinter para diseñar e implementar nuestra aplicación web.
  2. Lo más importante de este proyecto es que montaremos un flujo de datos desde cero, usando Python y SQL.
  3. Automatizaremos varios procesos.

Librería Tkinter¶

def dar_alta():
    # Creamos una nueva ventana y la denominamos vc_graficar
    # como no es una ventana principal, pondremos Toplevel()
    vc_graficar = tk.Toplevel()
    vc_graficar.geometry('500x350')
    
    # ------------------------------------------------------

    # Etiquetas
    etiqueta_title = ttk.Label(vc_graficar, text=' REGISTROS DE NUEVOS CLIENTES\n ', 
                               font=('Helvetica', 12), foreground='#000080')
    etiqueta_title.grid(row=0, column=1, sticky='NSWE')
    etiqueta_nombre = ttk.Label(vc_graficar, text='Nombre: ')
    etiqueta_nombre.grid(row=1, column=0, sticky='NSWE')
    etiqueta_edad = ttk.Label(vc_graficar, text='Edad: ')
    etiqueta_edad.grid(row=2, column=0, sticky='NSWE')
    etiqueta_sexo = ttk.Label(vc_graficar, text='Sexo: ')
    etiqueta_sexo.grid(row=3, column=0, sticky='NSWE')
    etiqueta_telefono = ttk.Label(vc_graficar, text='Telefono: ')
    etiqueta_telefono.grid(row=4, column=0, sticky='NSWE')
    etiqueta_pass = ttk.Label(vc_graficar, text='Password: ')
    etiqueta_pass.grid(row=5, column=0, sticky='NSWE')
    
    # ------------------------------------------------------

    # La caja de texto se mostrara en esta nueva ventana
    caja_nombre = ttk.Entry(vc_graficar, width=15)
    # Posicion de la caja de texto
    #                                                           pequenio margen
    #                                                           a la caja de texto
    n = 40
    caja_nombre.grid(row=1, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)
    caja_edad = ttk.Entry(vc_graficar, width=n)
    caja_edad.grid(row=2, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)
    # lo que hace combobox es generar una lista desplegable con los valores que le pasamos
    # en formato de una lista
    caja_sexo = ttk.Combobox(vc_graficar, state="readonly",
                             values=["M", "F", "O"])
    caja_sexo.grid(row=3, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)
    caja_telefono = ttk.Entry(vc_graficar, width=n)
    caja_telefono.grid(row=4, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)
    caja_pass = ttk.Entry(vc_graficar, width=n)
    caja_pass.grid(row=5, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)
    
    # ------------------------------------------------------

    # Ahora definiremos las funciones que realizaran
    # las acciones de nuestros botones.
    def subir_cliente():
        """ Lo que hace esta función es subir la informacion del cliente nuevo a nuestra base de datos
        """
        while True:
            # Realizamos el manejo de errores
            if caja_nombre.get() == '' or caja_edad.get() == '' or caja_sexo.get() == '' or caja_telefono.get() == '' or caja_pass.get() == '' or len(caja_pass.get()) != 4:
                messagebox.showerror('Error', 'Datos erróneos o longitud incorrecta del password')
                break
            try:
                int(caja_edad.get())
            except:
                messagebox.showerror('Error', 'Edad debe ser un número entero')
                break
               
            # info calculada: veamos en que dia estamos subiendo al cliente
            fecha_inicio = dt.datetime.now().strftime('%Y-%m-%d')
            # le damos 30 dias como su duracion de la membresia
            fecha_fin = (dt.datetime.now() + dt.timedelta(days=30)).strftime('%Y-%m-%d')
            estado = 'Activo'
            contador = 0
            
            # Conexion a la bbdd
            conn = sql.connect('clientes_db.db')
            c = conn.cursor()
            
            # info a subir
            query = f"""INSERT INTO clientes VALUES('{caja_nombre.get()}', 
            {int(caja_edad.get())}, '{caja_sexo.get()}', 
            '{caja_telefono.get()}', '{caja_pass.get()}', 
            '{fecha_inicio}', '{fecha_fin}',
            '{estado}', {contador}, '1900-00-00')
            """

            #try:
            c.execute(query)
            #except:
                #messagebox.showerror('Error', 'Password ya existente')
                #break
            conn.commit()
            conn.close()
            
            # Mensaje de exito
            msj = f'Usuario {caja_nombre.get()} subido a la base de datos'
            # mostramos el mensaje en un messagebox
            messagebox.showinfo(message=msj, title='Mensaje', parent=vc_graficar)
            break
    
    def clean():
        """ Funcion para limpiar la informacion que se encuentra en las cajas de texto
        """
        caja_nombre.delete(0, tk.END)
        caja_edad.delete(0, tk.END)
        caja_sexo.delete(0, tk.END)
        caja_telefono.delete(0, tk.END)
        caja_pass.delete(0, tk.END)
         
    # Botones:
    boton_sub_cliente = ttk.Button(vc_graficar, text='Subir cliente', command=subir_cliente)
    # # Posicion de este boton
    boton_sub_cliente.grid(row=6, column=1)
    
    # Boton para limpiar
    boton_sub_clean = ttk.Button(vc_graficar, text='Limpiar', command=clean)
    boton_sub_clean.grid(row=8, column=1)
    
    # Boton para salir
    boton_sub_salir = ttk.Button(vc_graficar, text='Regresar', command=vc_graficar.destroy)
    boton_sub_salir.grid(row=9, column=1)

def acceder():
    """
    Funcion para realizar el ingreso al gimnasio
    """
     
    # Ventana nueva
    ventana_m = tk.Toplevel()
    ventana_m.geometry('700x700')
    ventana_m.title('Acceso')
    
    # Etiquetas
    etiqueta_passw = ttk.Label(ventana_m, text='Password: ')
    etiqueta_passw.grid(row=0, column=0, sticky='NSWE')
    caja_passw = ttk.Entry(ventana_m, width=20)
    caja_passw.grid(row=0, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)
    
    def ingreso():
        # Obtenemos el password para realizar la consulta a la base de datos
        password = caja_passw.get()
        conn = sql.connect('clientes_db.db')
        query = f"""
        SELECT * FROM clientes WHERE password = '{password}'
        """
        # Ejecutamos y guardamos
        df = pd.read_sql_query(query, conn)
        conn.close()
        
        while True:
            if df.empty:
                messagebox.showerror('Error', 'Password incorrecto o no existe')
                break
            else:
                # concatenamos la informacion
                msj = f'Nombre: {df.iloc[0,0]}\
                    \n\nFecha de inicio: {df.iloc[0,5]}\
                    \n\nFecha de fin: {df.iloc[0,6]}\
                    \n\nEstado: {df.iloc[0,7]}'
                    
                # mostramos el mensaje en una etiqueta
                etiqueta_info = ttk.Label(ventana_m, text=msj)
                etiqueta_info.grid(row=3, column=1, sticky='NSWE')
                break
    
    boton1 = ttk.Button(ventana_m, text='Acceder', command=ingreso)
    boton1.grid(row=1, column=1, sticky='NSWE')

menu = tk.Menu(menu_config)

# Linea horizontal separadora
menu.add_separator()

# La funcion destroy() nos permite cerrar la ventana
menu.add_command(label = 'Salir', command=ventana.destroy)

# add_cascade() -> nos permite agregar las opciones del menu

menu_config.add_cascade(label='Menú', menu = menu)
