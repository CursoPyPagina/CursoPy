Clase 6: Desarrollo de aplicación para un gimnasio¶

Resumen parcial de códigos¶

  • ventana = tk.Tk(): creas tu ventana
  • ventana.geometry(): configuración del tamaño de la ventana
  • ventana.title(): configurar un título a la ventana
  • ttk: es parte de la librería de tkinter que nos permite colocar herramientas dentro de nuestra ventana.
  • ttk.Label(): colocar etiquetas en nuestra venta --> .grid(x, y): nos permite colocar en el lugar coordenada la ubicación de la etiqueta de texto.
  • ttk.Entry(): nos permite colocar cajas de texto.
  • ventana.mainloop(): ejecuta en bucle nuestra aplicación o interfaz.

def ingresar():
    """ Comando asociado a boton_entrar_login.
    * Nos conectaremos a la bbdd para traernos los passwords
    registrados.
    """
    # Parte 1: traernos la lista con los passwords
    con = sql.connect("Gym_pym.db")
    df = pd.read_sql_query("SELECT * FROM Administradores", con)
    con.close()
    passwords_lista = list(df['passwords'])
    # Parte 2: traernos ese password de la caja caja_etiqueta_password
    # dada una caja de texto, puedes extraer la informacion de ahi
    # con el metodo get()
    password_user = caja_etiqueta_password.get()
    # Parte 3: validar que el password sea correcto
    if password_user in passwords_lista:
        # en caso de que el password este en la lista permita
        # daremos acceso
        # ------------------booleano--------------------
        # yes ---> True
        # no  ---> False
        acceso = messagebox.askyesno(message='¿Deseas continuar?')

        if acceso:
            # si el administrador le da que si, entonces abriremos la 
            # app del gym (continuacion)
            login.destroy()
        else:
            # si le da en No el admin, entonces cerraremos la aplicacion
            login.destroy()
    else:
        # caso contrario, no daremos acceso
        messagebox.showerror('Contraseña incorrecta, intente de nuevo')

def subir_cliente():
    """ DOCUMENTACIÓN:
    Lo que hace esta función es subir la informacion del cliente nuevo a nuestra base de datos.
    """
    # Implementamos un bucle while para que la información se suba hasta que todo este correcto
    # Una vez que la info este correcta, detendremos el bucle while con la instruccion break
    while True:
        # MANEJO DE ERRORES
        # 1. Si alguna caja esdta vacia, dentendremos el programa y mandaremos un messagebox 
        if caja_nombre.get() == '' or caja_edad.get() == '' or caja_sexo.get() == '' or caja_telefono.get() == '' or caja_pass.get() == '' or len(caja_pass.get()) != 4:
            messagebox.showerror('Error', 'Datos erróneos o longitud incorrecta del password')
            break
        # 2. Exigimos que la edad sea un numero entero, recordemos que si le pasamos un string numerico
        # con un numero que NO es entero, entonces la funcion int() arrojara error, por ello aprovecharemos eso:
        try:
            int(caja_edad.get())
        except:
            messagebox.showerror('Error', 'Edad debe ser un número entero')
            break
           
        # info calculada: veamos en que dia estamos subiendo al cliente
        fecha_inicio = dt.datetime.now().strftime('%Y-%m-%d')
        # le damos 30 dias como su duracion de la membresia
        fecha_fin = (dt.datetime.now() + dt.timedelta(days=30)).strftime('%Y-%m-%d')
        estado = 'Activo'
        contador = 0
        
        # Conexion a la bbdd
        conn = sql.connect('clientes_db.db')
        c = conn.cursor()
        
        # info a subir 
        # OJO 👁️: no hemos definido esta tabla, lo haremos mas adelante
        query = f"""
            INSERT INTO 
                clientes 
            VALUES(
            '{caja_nombre.get()}', 
            {int(caja_edad.get())},
            '{caja_sexo.get()}', 
            '{caja_telefono.get()}',
            '{caja_pass.get()}', 
            '{fecha_inicio}',
            '{fecha_fin}',
            '{estado}', 
            {contador}
            )
        """

        # la insercion de arriba solo nos debe dar un posible error y eso tiene que ver con el password
        # Para ello, cuando definamos la tabla de SQL diremos que el password es una llave primaria
        # (mas adelante lo explicaremos, pero basicamente sirve para no meter duplicados a nuestra
        # base de datos).
        try:
        c.execute(query)
        except:
            messagebox.showerror('Error', 'Password ya existente')
            break
        conn.commit()
        conn.close()
        
        # Mensaje de exito
        msj = f'Usuario {caja_nombre.get()} subido a la base de datos'
        # mostramos el mensaje en un messagebox
        messagebox.showinfo(message=msj, title='Mensaje', parent=dar_alta)
        break

def acceder():
    """ DOCUMENTACIÓN:
    Funcion para realizar el ingreso al gimnasio
    """
     
    # Ventana nueva
    ingreso_v = tk.Toplevel()
    ingreso_v.geometry('700x700')
    ingreso_v.title('Acceso')
    
    # Etiquetas
    etiqueta_passw = ttk.Label(ingreso_v, text='Password: ')
    etiqueta_passw.grid(row=0, column=0, sticky='NSWE')
    caja_passw = ttk.Entry(ingreso_v, width=20)
    caja_passw.grid(row=0, column=1, sticky='NSWE', columnspan=2, padx=5, pady=5)

    # Comando para el boton
    def ingreso():
        # Obtenemos el password para realizar la consulta a la base de datos
        password = caja_passw.get()
        conn = sql.connect('Gym_pym.db.db')
        query = f"""
        SELECT * FROM clientes WHERE password = '{password}'
        """
        # Ejecutamos y guardamos
        df = pd.read_sql_query(query, conn)
        conn.close()
        
        while True:
            # si la consulta es vacia significa que el usuario no esta registrado
            if df.empty:
                messagebox.showerror('Error', 'Password incorrecto o no existe')
                break
            else:
                # concatenamos la informacion
                msj = f'Nombre: {df.iloc[0,0]}\
                    \n\nFecha de inicio: {df.iloc[0,5]}\
                    \n\nFecha de fin: {df.iloc[0,6]}\
                    \n\nEstado: {df.iloc[0,7]}'
                    
                # mostramos el mensaje en una etiqueta
                etiqueta_info = ttk.Label(ingreso_v, text=msj)
                etiqueta_info.grid(row=3, column=1, sticky='NSWE')
                break
    
    boton1 = ttk.Button(ingreso_v, text='Acceder', command=ingreso)
    boton1.grid(row=1, column=1, sticky='NSWE')
